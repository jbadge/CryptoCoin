let collectedData: Record<string, any[]> = {}
let collectedCount = 0

export function resetCollector() {
  collectedData = {}
  collectedCount = 0
}

export function addEntryForId(id: string, entries: any[], totalIds: number) {
  collectedData[id] = entries
  collectedCount++
  console.log(`Collected ${collectedCount} / ${totalIds}`)

  if (collectedCount === totalIds) {
    const output = Object.entries(collectedData).map(([id, entries]) => ({
      id,
      entries,
    }))
    const blob = new Blob([JSON.stringify(output, null, 2)], {
      type: 'application/json',
    })
    const url = URL.createObjectURL(blob)
    const a = document.createElement('a')
    a.href = url
    a.download = 'coin_history.json'
    a.click()
    URL.revokeObjectURL(url)
  }
}

// Fetch historical data for one coin
export async function fetchHistory(id: string) {
  try {
    const now = Date.now()
    const dayMs = 24 * 60 * 60 * 1000
    // Example: fetch history for the past 30 days
    const start = now - 30 * dayMs
    const end = now

    const response = await fetch(
      `https://rest.coincap.io/v3/assets/${id}/history?interval=d1&start=${start}&end=${end}&apiKey=${
        import.meta.env.VITE_API_KEY
      }`
    )
    if (response.ok) {
      const { data } = await response.json()
      return data
    } else {
      console.warn(`Failed to fetch history for ${id}`)
      return null
    }
  } catch (error) {
    console.error('Error fetching history for', id, error)
    return null
  }
}

////////////////
export async function slowFetchHistory(coins: Coins[]) {
  for (let i = 0; i < coins.length; i++) {
    const coin = coins[i]
    try {
      const data = await fetchHistory(coin.id)
      if (data) {
        addEntryForId(coin.id, data, coins.length)
      }
    } catch (err) {
      console.error(`Error fetching ${coin.id}`, err)
    }
    await new Promise((r) => setTimeout(r, 110))
  }
}
